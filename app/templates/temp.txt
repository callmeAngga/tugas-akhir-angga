    const USABLE_FRACTION = 0.70;

    // ---- Inisialisasi Peta ----
    const map = L.map('map', { zoomControl: true, preferCanvas: false })
        .setView([-6.2, 106.816], 12);

    const osm = L.tileLayer('https://{s}.tile.openstreetmap.org/{z}/{x}/{y}.png', {
        maxZoom: 20, attribution: '© OpenStreetMap'
    }).addTo(map);
    const sat = L.tileLayer(
        'https://server.arcgisonline.com/ArcGIS/rest/services/World_Imagery/MapServer/tile/{z}/{y}/{x}',

    // ---- Perbaikan tile blank/hitam: paksa recalculed ukuran kontainer ----
    function refreshMap() {
        map.invalidateSize({ animate: false });
    }
    // Panggil setelah layout & font stabil, dan tiap kali kontainer berubah ukuran.
    window.addEventListener('load', () => { refreshMap(); setTimeout(refreshMap, 250); });
    window.addEventListener('resize', refreshMap);
    if (document.fonts && document.fonts.ready) {
        document.fonts.ready.then(refreshMap);
    }
    if (window.ResizeObserver) {
        new ResizeObserver(refreshMap).observe(document.getElementById('map'));
    }
    map.whenReady(() => setTimeout(refreshMap, 100));

    // ---- Alat Gambar ----
    const drawn = new L.FeatureGroup();
    map.addLayer(drawn);
    const drawControl = new L.Control.Draw({
        position: 'topleft',
        edit: { featureGroup: drawn },
        draw: {
            rectangle: { shapeOptions: { color: '#2f81f7' } },
            polygon: { shapeOptions: { color: '#2f81f7' } },
            polyline: false, circle: false, marker: false, circlemarker: false
        }
    });
    map.addControl(drawControl);

    // State lokasi & area
    let currentLat = null, currentLon = null, currentArea = 0;

    function updateArea(layer) {
        const latlngs = layer.getLatLngs()[0];
        const area = L.GeometryUtil.geodesicArea(latlngs);
        const usable = area * USABLE_FRACTION;
        const c = layer.getBounds().getCenter();

        currentArea = area;
        currentLat = c.lat;
        currentLon = c.lng;

        document.getElementById('roofArea').textContent = Math.round(area).toLocaleString('id-ID') + ' m²';
        document.getElementById('usableArea').textContent = Math.round(usable).toLocaleString('id-ID') + ' m²';
        updateSystem(area);
        fetchRecommendation(c.lat, c.lng);
    }

    map.on(L.Draw.Event.CREATED, e => { drawn.clearLayers(); drawn.addLayer(e.layer); updateArea(e.layer); });
    map.on(L.Draw.Event.EDITED, e => e.layers.eachLayer(l => updateArea(l)));

    // ---- Logika Modul PV (termasuk kustom) ----
    function getModuleParams() {
        const sel = document.getElementById('pvModule');
        if (sel.value === 'custom') {
            return {
                watt: parseFloat(document.getElementById('customWatt').value) || 0,
                area: parseFloat(document.getElementById('customArea').value) || 0,
                name: document.getElementById('customName').value.trim() || 'Modul Kustom',
            };
        }
        const opt = sel.selectedOptions[0];
        return {
            watt: parseFloat(opt.dataset.watt),
            area: parseFloat(opt.dataset.area),
            name: opt.textContent.trim(),
        };
    }

    function updateSystem(area) {
        const m = getModuleParams();
        const usable = area * USABLE_FRACTION;
        const n = m.area > 0 ? Math.floor(usable / m.area) : 0;
        const cap = n * m.watt / 1000;

        document.getElementById('sumPanels').textContent = n.toLocaleString('id-ID');
        document.getElementById('sumCapacity').textContent = cap.toFixed(2) + ' kWp';
        document.getElementById('moduleSpec').value =
            m.watt > 0 ? m.watt + ' Wp · ' + m.area + ' m²/panel' : '—';
    }

    // Tampilkan/sembunyikan form kustom
    document.getElementById('pvModule').addEventListener('change', function () {
        const box = document.getElementById('customModuleBox');
        if (this.value === 'custom') {
            box.classList.add('visible');
        } else {
            box.classList.remove('visible');
        }
        updateSystem(currentArea);
    });

    // Update preview saat input kustom berubah
    ['customWatt', 'customArea', 'customName'].forEach(id => {
        document.getElementById(id).addEventListener('input', () => updateSystem(currentArea));

    // ---- Rekomendasi Orientasi Otomatis ----
    async function fetchRecommendation(lat, lon) {
        document.getElementById('runBtn').disabled = true;
        const box = document.getElementById('recoBox');
        box.classList.add('pending');
        document.getElementById('recoMethod').textContent = 'Menghitung orientasi optimal…';
        try {
            const res = await fetch('/api/recommend', {
                method: 'POST',
                headers: { 'Content-Type': 'application/json' },
                body: JSON.stringify({ lat, lon })
            });
            const d = await res.json();
            if (d.error) throw new Error(d.error);

            document.getElementById('recoTilt').textContent = d.tilt + '°';
            document.getElementById('recoAzimuth').textContent = d.azimuth + '° · ' + d.azimuth_label;
            document.getElementById('recoMethod').textContent =
                'Orientasi optimal berbasis fisika (transposisi Perez), konsisten dengan metodologi skripsi.';
            box.classList.remove('pending');

            if (document.getElementById('useReco').checked) {
                document.getElementById('tilt').value = d.tilt;
                document.getElementById('azimuth').value = d.azimuth;
            }
            box.dataset.tilt = d.tilt;
            box.dataset.azimuth = d.azimuth;
        } catch (err) {
            document.getElementById('recoMethod').textContent = 'Rekomendasi gagal: ' + err.message;
        } finally {
            document.getElementById('runBtn').disabled = false;
        }
    }

    // Toggle manual override
    document.getElementById('useReco').addEventListener('change', e => {
        const manual = document.getElementById('manualOrientation');
        manual.style.display = e.target.checked ? 'none' : 'grid';
        if (e.target.checked && document.getElementById('recoBox').dataset.tilt) {
            document.getElementById('tilt').value = document.getElementById('recoBox').dataset.tilt;
            document.getElementById('azimuth').value = document.getElementById('recoBox').dataset.azimuth;
        }
    });

    // ---- Pencarian Lokasi ----
    async function doSearch() {
        const q = document.getElementById('searchInput').value.trim();
        if (!q) return;
        const url = 'https://nominatim.openstreetmap.org/search?format=json&limit=1&q=' + encodeURIComponent(q);
        try {
            const res = await fetch(url, { headers: { 'Accept-Language': 'id' } });
            const data = await res.json();
            if (data.length) {
                map.setView([parseFloat(data[0].lat), parseFloat(data[0].lon)], 18);
                setTimeout(refreshMap, 300);
            } else {
                alert('Lokasi tidak ditemukan.');
            }
        } catch { alert('Pencarian gagal. Coba lagi.'); }
    }
    document.getElementById('searchBtn').addEventListener('click', doSearch);
    document.getElementById('searchInput').addEventListener('keydown', e => {
        if (e.key === 'Enter') { e.preventDefault(); doSearch(); }
    });

    const fmt = n => Number(n).toLocaleString('id-ID', { maximumFractionDigits: 0 });

            pv_module: document.getElementById('pvModule').value,
            custom_watt: m.watt,
            custom_area: m.area,
            custom_name: m.name,
            performance_ratio: document.getElementById('performance_ratio').value,
            tarif_rp_kwh: document.getElementById('tarif_rp_kwh').value,
            capex_per_kwp: document.getElementById('capex_per_kwp').value,

    // ---- Grafik GHI Prediksi (Chart.js) ----
    let ghiChart = null;
    function renderGhiChart(profile) {
        if (!profile || !profile.values) return;
        const ctx = document.getElementById('ghiChart').getContext('2d');

        // Label sumbu-x hanya di awal tiap bulan agar tidak padat.
        const monthNames = ['Jan', 'Feb', 'Mar', 'Apr', 'Mei', 'Jun',
            'Jul', 'Agu', 'Sep', 'Okt', 'Nov', 'Des'];
        const labels = profile.labels.map(d => {
            const dt = new Date(d);
            return dt.getDate() === 1 ? monthNames[dt.getMonth()] : '';
        });

        const grad = ctx.createLinearGradient(0, 0, 0, 320);
        grad.addColorStop(0, 'rgba(0,212,255,.35)');
        grad.addColorStop(1, 'rgba(0,212,255,0)');

        if (ghiChart) ghiChart.destroy();
        ghiChart = new Chart(ctx, {
            type: 'line',
            data: {
                labels: labels,
                datasets: [{
                    label: 'GHI (kWh/m²/hari)',
                    data: profile.values,
                    borderColor: '#00d4ff',
                    backgroundColor: grad,
                    borderWidth: 1.6,
                    pointRadius: 0,
                    pointHoverRadius: 4,
                    tension: 0.3,
                    fill: true,
                }]
            },
            options: {
                responsive: true,
                maintainAspectRatio: false,
                interaction: { mode: 'index', intersect: false },
                plugins: {
                    legend: { display: false },
                    tooltip: {
                        callbacks: {
                            title: items => 'Hari ke-' + (items[0].dataIndex + 1),
                            label: item => item.parsed.y.toFixed(2) + ' kWh/m²/hari',
                        }
                    }
                },
                scales: {
                    x: {
                        grid: { color: 'rgba(255,255,255,.05)' },
                        ticks: { color: '#8b98a9', autoSkip: false, maxRotation: 0 },
                    },
                    y: {
                        beginAtZero: true,
                        grid: { color: 'rgba(255,255,255,.06)' },
                        ticks: { color: '#8b98a9' },
                        title: { display: true, text: 'GHI (kWh/m²/hari)', color: '#8b98a9' },
                    }
                }
            }
        });

        document.getElementById('chartSub').textContent =
            'Forecast harian 2026 · total ' +
            fmt(profile.annual_kwh_m2) + ' kWh/m²/thn';
    }
